# Creating a Plugin

Use `@setu/plugin-sdk` to author custom plugins.

## Example Plugin Code

```typescript
import { GatewayPlugin, RequestContext } from "@setu/plugin-sdk";

export class CustomHeaderPlugin implements GatewayPlugin {
  name = "custom-header-plugin";

  async onRequest(context: RequestContext): Promise<void> {
    context.headers["X-Custom-Gateway-Header"] = "enabled";
  }
}
```
