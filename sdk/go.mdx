# Go SDK

Install the Go client package:

```bash
go get github.com/setu-gateway/sdk-go
```

## Quick Example

```go
package main

import (
    "context"
    "fmt"
    setu "github.com/setu-gateway/sdk-go"
)

func main() {
    client := setu.NewClient("http://localhost:8000/v1", "your-api-key")
    resp, err := client.CreateChatCompletion(context.Background(), setu.ChatRequest{
        Model: "gpt-4o",
        Messages: []setu.Message{{Role: "user", Content: "Hello"}},
    })
    if err != nil {
        panic(err)
    }
    fmt.Println(resp.Choices[0].Message.Content)
}
```
